// Type environment for typechecking the React Native hook-gen prompt's code
// fences that import @qontinui/ui-bridge-native (hook-gen-prompt-builder.test.ts).
// It mirrors, from ui-bridge d84bfa6, the public ui-bridge-native types those
// examples use: useUIElement and its option union, useUIComponent and
// ComponentActionDef, useRouteAwareness and the Expo route helpers, State /
// TransitionTo, and the provider's UndoTracker. NativeStandardAction and
// NativeCustomAction are simplified stand-ins (no example uses them), and the
// react / react-native / expo-router declarations are the minimum the
// examples touch, not those packages' real types. A plain-text fixture, not a
// .d.ts, so the app's own tsc never loads these ambient modules over the real
// @types/react.

declare namespace JSX {
  interface Element {}
  interface IntrinsicElements {
    [name: string]: any;
  }
  interface ElementChildrenAttribute {
    children: {};
  }
}

declare module 'react' {
  export type ReactNode = unknown;
  export function useState<S>(initial: S | (() => S)): [S, (next: S | ((prev: S) => S)) => void];
  export function useEffect(effect: () => void | (() => void), deps?: readonly unknown[]): void;
  export function useMemo<T>(factory: () => T, deps: readonly unknown[]): T;
}

declare module 'react-native' {
  export const Pressable: any;
  export const Text: any;
  export const TextInput: any;
  export const FlatList: any;
}

declare module 'expo-router' {
  export function useSegments(): string[];
}

declare module '@qontinui/ui-bridge-native' {
  import type { ReactNode } from 'react';

  // src/core/types.ts
  export type NativeElementType =
    | 'button'
    | 'input'
    | 'text'
    | 'view'
    | 'scroll'
    | 'list'
    | 'listItem'
    | 'switch'
    | 'checkbox'
    | 'radio'
    | 'image'
    | 'touchable'
    | 'pressable'
    | 'modal'
    | 'custom';
  export type NativeStandardAction = string;
  export type NativeCustomAction = unknown;
  export type IREffect = 'read' | 'write' | 'destructive';
  export interface RouteInfo {
    pattern?: string | null;
    patternSource?: 'router';
    params?: Record<string, string>;
    queryParams?: Record<string, string>;
    routeStack?: string[];
  }

  // src/react/useUIElement.ts
  type AnyHandler = (...args: any[]) => any;
  export type PressHandler = (event?: unknown) => void;
  export type PressNeedingNativeElementType = 'button' | 'pressable';
  export interface UseUIElementOptionsBase<T extends NativeElementType = NativeElementType> {
    id: string;
    type?: T;
    label?: string;
    actions?: NativeStandardAction[];
    customActions?: Record<string, NativeCustomAction>;
    value?: string;
    autoRegister?: boolean;
  }
  export interface UseUIElementOptionsHandlersRequired<
    T extends PressNeedingNativeElementType = PressNeedingNativeElementType,
  > extends UseUIElementOptionsBase<T> {
    handlers: { onPress: PressHandler; [key: string]: AnyHandler | undefined };
  }
  export interface UseUIElementOptionsHandlersOptional<
    T extends NativeElementType = NativeElementType,
  > extends UseUIElementOptionsBase<T> {
    handlers?: Record<string, AnyHandler | undefined>;
  }
  export type UseUIElementOptions<T extends NativeElementType = NativeElementType> =
    T extends PressNeedingNativeElementType
      ? UseUIElementOptionsHandlersRequired<T>
      : UseUIElementOptionsHandlersOptional<T>;
  export interface UIBridgeProps {
    testID: string;
    accessibilityLabel?: string;
  }
  export interface UseUIElementReturn {
    ref: { current: unknown };
    onLayout: (event: {
      nativeEvent: { layout: { x: number; y: number; width: number; height: number } };
    }) => void;
    bridgeProps: UIBridgeProps;
    registered: boolean;
  }
  export function useUIElement<T extends NativeElementType = NativeElementType>(
    options: UseUIElementOptions<T>,
  ): UseUIElementReturn;

  // src/react/useUIComponent.ts
  export interface ComponentActionDef<TParams = unknown, TResult = unknown> {
    id: string;
    label?: string;
    description?: string;
    paramSchema?: Record<string, unknown>;
    effect?: IREffect;
    handler: (params?: TParams, options?: { signal?: AbortSignal }) => TResult | Promise<TResult>;
  }
  export interface UseUIComponentOptions {
    id: string;
    name: string;
    description?: string;
    actions?: ComponentActionDef[];
    elementIds?: string[];
    autoRegister?: boolean;
  }
  export interface UseUIComponentReturn {
    registered: boolean;
    executeAction: <TParams = unknown, TResult = unknown>(
      actionId: string,
      params?: TParams,
    ) => Promise<TResult>;
  }
  export function useUIComponent(options: UseUIComponentOptions): UseUIComponentReturn;

  // src/react/useRouteAwareness.ts + src/react/routePattern.ts
  export interface UseRouteAwarenessOptions {
    unmatched?: boolean;
  }
  export function useRouteAwareness(info: RouteInfo, options?: UseRouteAwarenessOptions): void;
  export function expoRouterPattern(segments: readonly string[]): string | null;
  export function expoRouterRouteStack(segments: readonly string[]): string[] | undefined;
  export function isExpoRouterUnmatched(segments: readonly string[]): boolean;

  // src/react/StateAnnotation.tsx
  export interface StateAnnotationRequiredElement {
    role?: string;
    text?: string;
    testID?: string;
    [key: string]: unknown;
  }
  export interface StateProps {
    id: string;
    name?: string;
    description?: string;
    requiredElements?: StateAnnotationRequiredElement[];
    blocking?: boolean;
    children?: ReactNode;
    [key: string]: unknown;
  }
  export interface TransitionToProps {
    id: string;
    name?: string;
    fromStates?: string[];
    activateStates?: string[];
    exitStates?: string[];
    effect?: string;
    description?: string;
    children?: ReactNode;
    [key: string]: unknown;
  }
  export function State(props: StateProps): JSX.Element;
  export function TransitionTo(props: TransitionToProps): JSX.Element;

  // src/undo + src/react/UIBridgeNativeProvider.tsx
  export interface DeclaredUndoState {
    canUndo: boolean;
    canRedo: boolean;
    undoDescription?: string;
    redoDescription?: string;
    undoDepth?: number;
    redoDepth?: number;
  }
  export class UndoTracker {
    setDeclaredState(state: DeclaredUndoState): void;
  }
  export interface UIBridgeNativeContextValue {
    undoTracker: UndoTracker;
  }
  export function useUIBridgeNative(): UIBridgeNativeContextValue;
}

// App-side names the examples assume exist in the app being integrated.
declare function usePrompts(): { data: { id: string }[] | undefined; refetch(): void };
declare const router: { push(path: string): void };
declare function FilterSheet(props: { onClose(): void }): JSX.Element;
